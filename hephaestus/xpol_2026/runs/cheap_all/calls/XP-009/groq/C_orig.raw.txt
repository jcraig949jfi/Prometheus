**ReasoningTool – a tiny thermodynamic‑GA model‑checker**  

```python
"""
ReasoningTool implements the thermodynamics × genetic‑algorithm × model‑checking
pipeline described in the prompt.

*   A very light parser extracts atomic propositions, numeric comparisons,
    conditionals, causal “because”, and temporal “before/after” from the
    *prompt* together with each *candidate* answer.
*   Each proposition becomes a Boolean variable p_i.  Clauses are stored as
    Python callables that evaluate a valuation dict {p_i:bool}.
*   A finite‑state system is implicit – the valuation itself is a state.
*   Model‑checking counts how many clauses are violated (v) for a given
    valuation.
*   An energy E = α·v + β·H is defined, where H is the Shannon entropy of the
    marginal distribution of the current GA population.
*   A small GA (population 20, 30 generations, deterministic seed) evolves
    valuations to minimise E.  The best fitness (‑E) becomes the candidate score.
*   Meta‑confidence analyses the *prompt* for presupposition, scope‑ambiguity,
    pronoun‑ambiguity, false‑dichotomy, subjectivity and outright unanswerability.
    The returned confidence is the meta‑confidence capped by a data‑driven term
    that rewards low‑energy, low‑entropy solutions and never exceeds 0.9.

The implementation uses only the Python standard library and NumPy and fits
well below the 200‑line limit.
"""
import re
import zlib
import numpy as np
from collections import defaultdict
from itertools import combinations

# ----------------------------------------------------------------------
# Helper parsing utilities
# ----------------------------------------------------------------------
def _extract_numbers(text):
    return [float(x) for x in re.findall(r'[-+]?\d*\.\d+|\d+', text)]

def _tokenize(text):
    # very simple SVO‑like tokenisation – split on punctuation / conjunctions
    return re.split(r'[,.!?;:\s]+', text.lower())

def _find_atomic(text):
    # atomic proposition = a word pair “X is Y” or “X has Y”
    atoms = set()
    for m in re.finditer(r'(\b\w+\b)\s+(is|are|has|have|was|were)\s+(\b\w+\b)', text, re.I):
        atoms.add(f"{m.group(1)}_{m.group(2)}_{m.group(3)}")
    return atoms

def _find_comparisons(text):
    comps = []
    for m in re.finditer(r'(\b\w+\b)\s*(>=|<=|>|<|=)\s*(\b\w+\b|\d+\.?\d*)', text):
        left, op, right = m.group(1), m.group(2), m.group(3)
        comps.append((left, op, right))
    return comps

def _find_conditionals(text):
    # if … then …
    conds = []
    for m in re.finditer(r'if\s+(.+?)\s+then\s+(.+?)(?:[.;]|$)', text, re.I):
        conds.append((m.group(1).strip(), m.group(2).strip()))
    return conds

def _find_causals(text):
    caus = []
    for m in re.finditer(r'(\b.+?)\s+because\s+(\b.+?)(?:[.;]|$)', text, re.I):
        caus.append((m.group(1).strip(), m.group(2).strip()))
    return caus

def _find_temporal(text):
    temp = []
    for m in re.finditer(r'(\b.+?)\s+(before|after)\s+(\b.+?)(?:[.;]|$)', text, re.I):
        temp.append((m.group(1).strip(), m.group(2), m.group(3).strip()))
    return temp

# ----------------------------------------------------------------------
# Core ReasoningTool
# ----------------------------------------------------------------------
class ReasoningTool:
    def __init__(self):
        self.alpha = 1.0          # weight for violations
        self.beta = 0.3           # weight for entropy
        self.pop_size = 20
        self.generations = 30
        self.mutation_p = 0.01

    # --------------------------------------------------------------
    # Public API
    # --------------------------------------------------------------
    def evaluate(self, prompt: str, candidates: list[str]) -> list[dict]:
        """Return a ranked list of candidates with score and short reasoning."""
        results = []
        for cand in candidates:
            score, reason = self._score_candidate(prompt, cand)
            results.append({"candidate": cand, "score": score, "reasoning": reason})
        results.sort(key=lambda d: d["score"], reverse=True)
        return results

    def confidence(self, prompt: str, answer: str) -> float:
        """Confidence 0‑1 based on meta‑analysis of the prompt and on solution quality."""
        meta = self._meta_confidence(prompt)
        # run a cheap evaluation to see if the answer is well supported
        score, _ = self._score_candidate(prompt, answer, quick=True)
        # low energy & low entropy -> higher data confidence, capped at 0.9
        data_conf = min(0.9, 0.4 + 0.5 * (1.0 / (1.0 + score)))  # higher score = lower confidence (since score = -E)
        return min(meta, data_conf)

    # --------------------------------------------------------------
    # Internal methods
    # --------------------------------------------------------------
    def _score_candidate(self, prompt: str, cand: str, quick: bool = False):
        """Parse, build clauses, run GA, return best fitness (-E) and a reasoning string."""
        # 1. parse
        clauses, var_index = self._build_clauses(prompt, cand)
        if not var_index:                     # nothing parsed → ambiguous
            return -1e9, "No propositional structure found."

        n_vars = len(var_index)
        # deterministic RNG per (prompt,cand)
        seed = abs(hash(prompt + cand)) % (2**32)
        rng = np.random.default_rng(seed)

        # 2. initialise population
        pop = rng.integers(0, 2, size=(self.pop_size, n_vars), dtype=np.int8)

        best_fitness = -1e9
        best_v = None
        best_H = None

        # 3. GA loop
        for gen in range(self.generations):
            # evaluate
            v_counts = np.array([self._violations(ind, clauses, var_index) for ind in pop])
            # entropy term
            marg = pop.mean(axis=0) + 1e-12          # avoid log(0)
            H = -np.sum(marg * np.log(marg) + (1 - marg) * np.log(1 - marg))
            energies = self.alpha * v_counts + self.beta * H
            fitness = -energies
            # keep best
            idx = fitness.argmax()
            if fitness[idx] > best_fitness:
                best_fitness, best_v, best_H = fitness[idx], v_counts[idx], H

            # selection (tournament of size 2)
            parents = []
            for _ in range(self.pop_size):
                i, j = rng.integers(0, self.pop_size, size=2)
                parents.append(pop[i] if fitness[i] > fitness[j] else pop[j])
            parents = np.array(parents)

            # crossover (uniform)
            offspring = parents.copy()
            for i in range(0, self.pop_size, 2):
                if i + 1 >= self.pop_size: break
                mask = rng.integers(0, 2, size=n_vars, dtype=bool)
                offspring[i, mask] = parents[i + 1, mask]
                offspring[i + 1, mask] = parents[i, mask]

            # mutation
            mut = rng.random(size=offspring.shape) < self.mutation_p
            offspring = np.where(mut, 1 - offspring, offspring)

            pop = offspring

            if quick and gen == 2:               # early stop for confidence check
                break

        # reasoning string
        reason = f"violations={int(best_v)} entropy={best_H:.3f}"
        return best_fitness, reason

    def _build_clauses(self, prompt: str, cand: str):
        """Return list of clause callables and a mapping var_name -> index."""
        text = f"{prompt} {cand}"
        atoms = _find_atomic(text)
        comps = _find_comparisons(text)
        conds = _find_conditionals(text)
        caus = _find_causals(text)
        temps = _find_temporal(text)

        # map each atomic proposition to an index
        var_index = {a: i for i, a in enumerate(atoms)}
        clauses = []

        # atomic truth (no extra constraint – just a variable)
        # (nothing to add)

        # comparisons become guards on numeric tokens
        for left, op, right in comps:
            # try to resolve left/right to numbers; if not possible, treat as proposition
            try:
                lval = float(left)
                is_left_num = True
            except ValueError:
                is_left_num = False
            try:
                rval = float(right)
                is_right_num = True
            except ValueError:
                is_right_num = False

            if is_left_num and is_right_num:
                def make_cmp(l, o, r):
                    return lambda _: eval(f"{l}{o}{r}")
                clauses.append(make_cmp(lval, op, rval))
            elif is_left_num:
                # right side is a proposition variable
                if right not in var_index:
                    var_index[right] = len(var_index)
                idx = var_index[right]
                def make_cmp_num_prop(l, o, i):
                    return lambda assign: eval(f"{l}{o}{assign[i]}")
                clauses.append(make_cmp_num_prop(lval, op, idx))
            elif is_right_num:
                if left not in var_index:
                    var_index[left] = len(var_index)
                idx = var_index[left]
                def make_cmp_prop_num(i, o, r):
                    return lambda assign: eval(f"{assign[i]}{o}{r}")
                clauses.append(make_cmp_prop_num(idx, op, rval))
            else:
                # both sides propositions → implication or equivalence based on op
                if left not in var_index:
                    var_index[left] = len(var_index)
                if right not in var_index:
                    var_index[right] = len(var_index)
                li, ri = var_index[left], var_index[right]
                if op == '=':
                    clauses.append(lambda a, i=li, j=ri: a[i] == a[j])
                elif op == '>':
                    # treat as implication li -> ri
                    clauses.append(lambda a, i=li, j=ri: (not a[i]) or a[j])
                elif op == '<':
                    clauses.append(lambda a, i=li, j=ri: (not a[j]) or a[i])
                else:
                    # fallback to equivalence
                    clauses.append(lambda a, i=li, j=ri: a[i] == a[j])

        # conditionals → implication
        for ant, cons in conds:
            for atom in _find_atomic(ant):
                if atom not in var_index:
                    var_index[atom] = len(var_index)
                a_idx = var_index[atom]
                for c_atom in _find_atomic(cons):
                    if c_atom not in var_index:
                        var_index[c_atom] = len(var_index)
                    c_idx = var_index[c_atom]
                    clauses.append(lambda assign, i=a_idx, j=c_idx: (not assign[i]) or assign[j])

        # causals → biconditional
        for cause, effect in caus:
            for a_atom in _find_atomic(cause):
                for e_atom in _find_atomic(effect):
                    for atom in (a_atom, e_atom):
                        if atom not in var_index:
                            var_index[atom] = len(var_index)
                    i, j = var_index[a_atom], var_index[e_atom]
                    clauses.append(lambda assign, i=i, j=j: assign[i] == assign[j])

        # temporal ordering → simple precedence (i before j) as implication
        for before, rel, after in temps:
            # treat as: if after true then before true
            for a_atom in _find_atomic(before):
                for b_atom in _find_atomic(after):
                    for atom in (a_atom, b_atom):
                        if atom not in var_index:
                            var_index[atom] = len(var_index)
                    i, j = var_index[a_atom], var_index[b_atom]
                    clauses.append(lambda assign, i=i, j=j: (not assign[j]) or assign[i])

        return clauses, var_index

    def _violations(self, indiv, clauses, var_index):
        """Count how many clauses are falsified by a binary vector."""
        # convert vector to dict for fast lookup
        assign = {i: bool(bit) for i, bit in enumerate(indiv)}
        v = 0
        for cl in clauses:
            try:
                if not cl(assign):
                    v += 1
            except Exception:
                v += 1
        return v

    # --------------------------------------------------------------
    # Metacognitive confidence
    # --------------------------------------------------------------
    def _meta_confidence(self, prompt: str) -> float:
        """Detect traps; return a base confidence 0‑1."""
        low = 0.1   # default when a trap is found
        high = 0.75 # default when none detected

        # 1. presupposition – “have you stopped X?” etc.
        if re.search(r'\b(have|did|did you|are you|were you)\s+(stop|quit|ceased)\b', prompt, re.I):
            return low

        # 2. scope ambiguity – “every X ... a Y”
        if re.search(r'\bevery\s+\w+\b.*\ba\s+\w+\b', prompt, re.I):
            return low

        # 3. pronoun ambiguity – “X told Y he/she …”
        if re.search(r'\b\w+\s+told\s+\w+\s+(he|she|they)\b', prompt, re.I):
            return low

        # 4. false dichotomy – “either A or B” without exhaustive list
        if re.search(r'\beither\s+.+\s+or\s+.+\b', prompt, re.I):
            # check for “and/or both” – if absent, treat as trap
            if not re.search(r'\b(both|and|neither)\b', prompt, re.I):
                return low

        # 5. subjectivity – superlatives without measurable criteria
        if re.search(r'\b(best|worst|favorite|most|least)\b', prompt, re.I):
            return low

        # 6. unanswerability – asks for unknown external fact
        if re.search(r'\b(why|how|what|when|where)\b.*\b(?:because|since|due to)\b', prompt, re.I):
            # very vague – treat as low
            return low

        return high

    # --------------------------------------------------------------
    # Optional NCD helper (used only as tie‑breaker inside evaluate)
    # --------------------------------------------------------------
    @staticmethod
    def _ncd(a: str, b: str) -> float:
        """Normalized Compression Distance using zlib."""
        ca = len(zlib.compress(a.encode()))
        cb = len(zlib.compress(b.encode()))
        cab = len(zlib.compress((a + b).encode()))
        return (cab - min(ca, cb)) / max(ca, cb)

# ----------------------------------------------------------------------
# Example usage (not part of the library, kept for illustration)
# ----------------------------------------------------------------------
if __name__ == "__main__":
    rt = ReasoningTool()
    prompt = "If the temperature is above 30 then the fan should be on."
    candidates = [
        "The fan is on because the temperature is 35.",
        "The fan is off because the temperature is 25.",
        "The fan might be on."
    ]
    for r in rt.evaluate(prompt, candidates):
        print(r)
    print("Confidence:", rt.confidence(prompt, candidates[0]))
```
