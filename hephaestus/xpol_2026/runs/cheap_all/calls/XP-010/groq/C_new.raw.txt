**ReasoningTool – a pure‑numpy reasoning evaluator**  

```python
"""
ReasoningTool implements the Morphogenesis × Spectral × Maximum‑Entropy
pipeline described in the prompt.

* Parsing – extracts simple subject‑predicate‑object triples and a few
  structural cues (negation, comparatives, conditionals, numbers).

* Graph construction – each unique token becomes a vertex; edges are added
  for the extracted triples and for adjacency in the original text.
  Edge weights are 1.0 for explicit relations and 0.1 for simple adjacency.

* Reaction‑diffusion – a Turing‑type update runs on the concentration vector
  `c`.  After a fixed number of iterations the pattern settles.

* Spectral fingerprint – a 1‑D FFT of the final concentrations yields a power
  spectrum `P`.  The mean `μ` and variance `σ²` of `P` are fed to a
  maximum‑entropy (Gaussian) model; the expected value `μ` is turned into a
  0‑100 score.

* Structural similarity – overlap of cue‑type tokens between prompt and a
  candidate contributes a second signal.

* Optional NCD – a tiny zlib‑based Normalised Compression Distance is used
  only as a tie‑breaker (≤ 15 % weight).

* Meta‑confidence – regex checks for presupposition, scope‑/pronoun‑
  ambiguity, false dichotomy, subjectivity and unanswerability.  When any
  pattern matches the confidence is forced below 0.3, otherwise it is the
  smaller of the meta‑confidence and the candidate‑score‑derived confidence.

The implementation is deterministic, uses only the standard library and
numpy, and fits well under the 200‑line limit.
"""

import re
import zlib
import numpy as np
from collections import defaultdict
from itertools import combinations

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _tokens(text):
    """Very light tokeniser – keeps numbers, words and punctuation."""
    return re.findall(r"\b\w+\.?\w*\b|[<>]=?|[^\s]", text.lower())

def _extract_triples(tokens):
    """Return a list of (subj, pred, obj) triples using a tiny heuristic."""
    triples = []
    # look for patterns: noun verb noun  (e.g. "rain causes flood")
    for i in range(len(tokens) - 2):
        if tokens[i].isalpha() and tokens[i+1].isalpha() and tokens[i+2].isalpha():
            triples.append((tokens[i], tokens[i+1], tokens[i+2]))
    return triples

def _structural_cues(text):
    """Detect cue‑type tokens used for the structural similarity component."""
    cues = {
        "neg": {"not", "no", "without", "never"},
        "comp": {"greater", "less", "more", "fewer", "as", "than"},
        "cond": {"if", "when", "unless", "then"},
        "num": set(re.findall(r"\b\d+(\.\d+)?\b", text)),
        "caus": {"cause", "leads", "result", "because"},
        "ord": {"first", "second", "subsequently", "finally"},
    }
    found = set()
    words = set(_tokens(text))
    for cat, words_set in cues.items():
        if words & words_set:
            found.add(cat)
    return found

def _ncd(a, b):
    """Normalised Compression Distance using zlib."""
    ca = len(zlib.compress(a.encode()))
    cb = len(zlib.compress(b.encode()))
    cab = len(zlib.compress((a + b).encode()))
    return (cab - min(ca, cb)) / max(ca, cb)

# ----------------------------------------------------------------------
# Core class
# ----------------------------------------------------------------------
class ReasoningTool:
    """Implements the morphogenesis‑spectral‑entropy reasoning evaluator."""

    def __init__(self):
        # diffusion / reaction hyper‑parameters (chosen empirically)
        self.D = 0.2          # diffusion coefficient
        self.R = 0.1          # reaction coefficient
        self.steps = 30       # diffusion iterations
        self.ncd_weight = 0.15
        self.struct_weight = 0.30
        self.diff_weight = 0.55

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------
    def evaluate(self, prompt: str, candidates: list[str]) -> list[dict]:
        """Score each candidate and return a ranked list of dicts."""
        prompt_cues = _structural_cues(prompt)
        results = []
        for cand in candidates:
            # 1. structural overlap
            cand_cues = _structural_cues(cand)
            struct_score = len(prompt_cues & cand_cues) / max(1, len(prompt_cues))

            # 2. diffusion‑spectral score
            diff_score = self._diffusion_score(prompt, cand)

            # 3. optional NCD (tiny tie‑breaker)
            ncd_score = 1 - _ncd(prompt, cand)   # higher = more similar
            ncd_score = max(0.0, min(1.0, ncd_score))

            # weighted combination
            total = (self.diff_weight * diff_score +
                     self.struct_weight * struct_score +
                     self.ncd_weight * ncd_score)

            # scale to 0‑100 and build output dict
            results.append({
                "candidate": cand,
                "score": round(total * 100, 2),
                "reasoning": (f"struct={struct_score:.2f}, "
                              f"diff={diff_score:.2f}, ncd={ncd_score:.2f}")
            })
        # rank descending by score
        return sorted(results, key=lambda x: x["score"], reverse=True)

    def confidence(self, prompt: str, answer: str) -> float:
        """
        Return a confidence 0‑1 based on prompt ambiguity (meta‑confidence)
        and the answer's own score.  Never exceeds 0.9 unless the answer is
        clearly computed.
        """
        meta = self._meta_confidence(prompt)
        # compute a raw confidence from the answer's own score (if we have it)
        # we reuse the diffusion score as a proxy for answer quality
        raw = self._diffusion_score(prompt, answer)
        # combine – the lower of the two dominates (honest uncertainty)
        conf = min(meta, raw)
        # cap at 0.9 for non‑definitive cases
        return min(conf, 0.9)

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _diffusion_score(self, prompt: str, candidate: str) -> float:
        """
        Build a graph from prompt + candidate, run reaction‑diffusion,
        perform a 1‑D FFT on the steady‑state concentrations and map the mean
        spectral power to a 0‑1 score.
        """
        # ----- graph construction ------------------------------------------------
        tokens = _tokens(prompt) + _tokens(candidate)
        uniq = list(dict.fromkeys(tokens))          # preserve order, drop dupes
        idx = {tok: i for i, tok in enumerate(uniq)}
        n = len(uniq)

        # adjacency matrix (weights)
        W = np.zeros((n, n), dtype=float)

        # explicit triples from both texts
        for txt in (prompt, candidate):
            toks = _tokens(txt)
            for s, p, o in _extract_triples(toks):
                if s in idx and o in idx:
                    i, j = idx[s], idx[o]
                    W[i, j] = W[j, i] = 1.0   # strong relation

        # simple adjacency (window size 2) as weak diffusion links
        for i in range(len(tokens) - 1):
            a, b = idx[tokens[i]], idx[tokens[i + 1]]
            if a != b:
                W[a, b] = W[b, a] = max(W[a, b], 0.1)

        # degree matrix and Laplacian
        deg = np.sum(W, axis=1)
        L = np.diag(deg) - W

        # ----- reaction‑diffusion -------------------------------------------------
        # initialise concentrations: 1 for tokens that appear in both texts,
        # 0 otherwise (simple proxy for "reference match")
        c = np.array([1.0 if tok in _tokens(prompt) and tok in _tokens(candidate)
                      else 0.0 for tok in uniq], dtype=float)

        for _ in range(self.steps):
            lap = -L @ c                     # Laplacian term (negative sign)
            reaction = -self.R * np.sum(W * (c[:, None] - c[None, :]), axis=1)
            c = c + self.D * lap + reaction
            # keep concentrations bounded
            c = np.clip(c, 0.0, 1.0)

        # ----- spectral analysis --------------------------------------------------
        fft_vals = np.fft.rfft(c)            # real FFT (efficient)
        power = np.abs(fft_vals) ** 2
        mu = power.mean()
        var = power.var() + 1e-9

        # maximum‑entropy mapping: expected value of Gaussian = mu
        # normalise mu to [0,1] using a simple logistic that also uses var
        score = 1.0 / (1.0 + np.exp(-(mu - mu) / np.sqrt(var)))   # =0.5
        # we want a monotonic function of mu; use a scaled logistic centred at
        # the median of power (approx mu).  A simpler linear normalisation works:
        score = mu / (mu + var)          # lies in (0,1)

        return float(score)

    # ------------------------------------------------------------------
    # Meta‑confidence detection
    # ------------------------------------------------------------------
    def _meta_confidence(self, prompt: str) -> float:
        """Detect ambiguous / presuppositional patterns and return a base confidence."""
        txt = prompt.lower()

        # 1. presupposition
        if re.search(r"\b(have you|did you|are you) (stopped|quit|ceased|given up) \w+", txt):
            return 0.2

        # 2. scope ambiguity – "every X ... a Y" (simple heuristic)
        if re.search(r"\bevery\b.*\ba\b", txt):
            return 0.25

        # 3. pronoun ambiguity – "X told Y he/she" without clarification
        if re.search(r"\b\w+ told \w+ (he|she|they)\b", txt):
            return 0.2

        # 4. false dichotomy – "either A or B" with no "both" or "neither"
        if re.search(r"\beither\b.*\bor\b", txt) and not re.search(r"\bboth\b|\bneither\b", txt):
            return 0.25

        # 5. subjectivity – superlatives without measurable criteria
        if re.search(r"\b(best|worst|favorite|most|least)\b", txt):
            return 0.3

        # 6. unanswerability – asks for unknown external data
        if re.search(r"\b(what|how|why|when|where)\b.*\b(is|are|was|were)\b.*\b\w+\b", txt) and \
           not re.search(r"\baccording to\b|\bgiven\b|\bprovided\b", txt):
            return 0.2

        # default – no trap detected
        return 0.8

# ----------------------------------------------------------------------
# Example usage (can be removed in production)
# ----------------------------------------------------------------------
if __name__ == "__main__":
    rt = ReasoningTool()
    prompt = "If the rain causes flooding, what happens when the river overflows?"
    cand = ["The flood worsens", "Water levels rise", "Nothing changes"]
    for r in rt.evaluate(prompt, cand):
        print(r)
    print("Confidence:", rt.confidence(prompt, cand[0]))
```
