**ReasoningTool – a tiny, deterministic reasoning‑evaluator**  

```python
"""
ReasoningTool implements the Information‑Theory × Mechanism‑Design × Proof‑Theory
pipeline described in the prompt together with a very small dynamical‑system
tracker.  For every candidate answer it

1. parses the prompt and the candidate into a proof‑net‑like hyper‑graph
   (regex‑driven extraction of atomic propositions, conditionals, comparatives,
   negations, causal cues and numeric constants);
2. normalises the graph by cut‑elimination style rewrites
   (double‑negation removal, modus‑ponens, transitive closure of ordering);
3. builds a predicate distribution and scores the candidate with the proper
   scoring rule  S = -KL(q‖p) + H(q)  where q comes from the prompt (reference)
   and p from the candidate;
4. runs a tiny state‑space dynamics: each predicate has a deterministic
   embedding; the state vector is updated by   s←tanh(s+embed)   for every
   premise in order.  The final state is recomputed on a few random
   permutations – low variance ⇒ high stability score;
5. combines the structural score and the dynamics stability (each 0‑1) and
   adds a tiny NCD‑based tiebreaker (≤15 % of the final weight).

The ``confidence`` method is *epistemically honest*: it inspects the prompt
for presuppositions, scope/pronoun ambiguity, false dichotomies,
subjectivity and outright unanswerability and returns a low confidence
whenever any of those patterns appear.  The confidence never exceeds 0.9
unless the question is clearly answerable and the dynamics are stable.

Only the Python standard library and NumPy are used; the implementation is
well under 200 lines and deterministic (fixed random seed for embeddings).
"""
import re
import zlib
import numpy as np
from itertools import permutations, islice
from collections import Counter, defaultdict

# ---------- helper utilities -------------------------------------------------
def _tokenize(text: str) -> str:
    """Normalise whitespace and lower‑case."""
    return re.sub(r'\s+', ' ', text.strip().lower())

def _kl_div(q, p):
    """KL(q‖p) with smoothing for zero entries."""
    eps = 1e-12
    q = np.asarray(q) + eps
    p = np.asarray(p) + eps
    return np.sum(q * np.log(q / p))

def _entropy(p):
    eps = 1e-12
    p = np.asarray(p) + eps
    return -np.sum(p * np.log(p))

def _ncd(a: str, b: str) -> float:
    """Normalised compression distance using zlib."""
    ca = len(zlib.compress(a.encode()))
    cb = len(zlib.compress(b.encode()))
    cab = len(zlib.compress((a + b).encode()))
    return (cab - min(ca, cb)) / max(ca, cb)


# ---------- core classes ------------------------------------------------------
class _HyperGraph:
    """Very small proof‑net‑like representation."""
    def __init__(self):
        self.nodes = []          # list of (pred, args) tuples
        self.edges = []          # list of (type, src_idx, tgt_idx)

    def add_node(self, pred, *args):
        self.nodes.append((pred, args))
        return len(self.nodes) - 1

    def add_edge(self, typ, src, tgt):
        self.edges.append((typ, src, tgt))


class ReasoningTool:
    """Implements the pipeline from the specification."""
    # fixed predicate dictionary – can be extended
    _PREDICATES = [
        'neg', 'cond', 'gt', 'lt', 'ge', 'le',
        'eq', 'before', 'after', 'because', 'leads_to',
        'numeric', 'quant_all', 'quant_some'
    ]

    def __init__(self):
        # deterministic embeddings for each predicate
        rng = np.random.RandomState(42)
        self._embed = {
            p: rng.normal(scale=0.1, size=10) for p in self._PREDICATES
        }

    # -------------------------------------------------------------------------
    # 1. PARSING → LOGICAL FORM
    # -------------------------------------------------------------------------
    def _parse(self, text: str) -> _HyperGraph:
        hg = _HyperGraph()
        txt = _tokenize(text)

        # 1) conditionals  "if A then B"
        for m in re.finditer(r'if (.+?) then (.+?)(?:[.;]|$)', txt):
            ant = m.group(1).strip()
            con = m.group(2).strip()
            a = hg.add_node('cond', ant)
            c = hg.add_node('cond', con)
            hg.add_edge('cond', a, c)

        # 2) comparatives  "x > 5", "x less than 3"
        comp_pat = r'(\w+)\s*(>=|<=|>|<|=)\s*([\d\.]+)'
        for m in re.finditer(comp_pat, txt):
            left, op, right = m.groups()
            pred = {'>': 'gt', '<': 'lt',
                    '>=': 'ge', '<=': 'le',
                    '=': 'eq'}[op]
            hg.add_node(pred, left, right)

        # 3) ordering cues  "A before B", "after C"
        for m in re.finditer(r'(\w+)\s+(before|after)\s+(\w+)', txt):
            a, rel, b = m.groups()
            pred = rel
            hg.add_node(pred, a, b)

        # 4) causal cues  "because", "leads to"
        for m in re.finditer(r'(\w+)\s+because\s+(\w+)', txt):
            a, b = m.groups()
            hg.add_node('because', a, b)
        for m in re.finditer(r'(\w+)\s+leads to\s+(\w+)', txt):
            a, b = m.groups()
            hg.add_node('leads_to', a, b)

        # 5) negations  "not X"
        for m in re.finditer(r'not (\w+)', txt):
            hg.add_node('neg', m.group(1))

        # 6) quantifiers
        for m in re.finditer(r'\b(all|every) (\w+)', txt):
            hg.add_node('quant_all', m.group(2))
        for m in re.finditer(r'\b(some|a few) (\w+)', txt):
            hg.add_node('quant_some', m.group(2))

        # 7) isolated numbers (treated as numeric facts)
        for m in re.finditer(r'\b([\d\.]+)\b', txt):
            hg.add_node('numeric', m.group(1))

        return hg

    # -------------------------------------------------------------------------
    # 2. NORMALISATION (CUT‑ELIMINATION‑STYLE)
    # -------------------------------------------------------------------------
    def _normalize(self, hg: _HyperGraph) -> _HyperGraph:
        # copy to avoid mutating original
        new_hg = _HyperGraph()
        new_hg.nodes = list(hg.nodes)
        new_hg.edges = list(hg.edges)

        # double‑negation removal
        changed = True
        while changed:
            changed = False
            for i, (pred, args) in enumerate(new_hg.nodes):
                if pred == 'neg' and args and isinstance(args[0], str):
                    inner = args[0]
                    # look for a neg node with same argument
                    for j, (p2, a2) in enumerate(new_hg.nodes):
                        if p2 == 'neg' and a2 and a2[0] == inner:
                            # replace both with inner (no neg)
                            new_hg.nodes[i] = ('pos', (inner,))
                            new_hg.nodes.pop(j)
                            changed = True
                            break
                if changed:
                    break

        # modus ponens on conditionals
        added = set()
        for typ, src, tgt in new_hg.edges:
            if typ == 'cond':
                antecedent = new_hg.nodes[src][1][0] if new_hg.nodes[src][1] else ''
                consequent = new_hg.nodes[tgt][1][0] if new_hg.nodes[tgt][1] else ''
                # if antecedent also appears as a node, we can add consequent
                for i, (p, a) in enumerate(new_hg.nodes):
                    if a and a[0] == antecedent and i not in added:
                        new_hg.add_node('derived', consequent)
                        added.add(i)

        # transitive closure for ordering (before/after)
        order = defaultdict(set)          # before relations
        for pred, args in new_hg.nodes:
            if pred in ('before', 'after') and len(args) == 2:
                a, b = args
                if pred == 'before':
                    order[a].add(b)
                else:                     # after => b before a
                    order[b].add(a)

        # Floyd‑Warshall style closure (small graph)
        for k in order:
            for i in order:
                if k in order[i]:
                    order[i].update(order[k])

        # add inferred before edges
        for a, targets in order.items():
            for b in targets:
                new_hg.add_node('before', a, b)

        return new_hg

    # -------------------------------------------------------------------------
    # 3. INFORMATION‑THEORETIC SCORING
    # -------------------------------------------------------------------------
    def _predicate_distribution(self, hg: _HyperGraph):
        cnt = Counter(p for p, _ in hg.nodes)
        total = sum(cnt.values())
        vec = np.array([cnt.get(p, 0) / total for p in self._PREDICATES])
        return vec

    def _structural_score(self, cand_hg: _HyperGraph, ref_hg: _HyperGraph) -> float:
        p = self._predicate_distribution(cand_hg)
        q = self._predicate_distribution(ref_hg)
        Hq = _entropy(q)
        KL = _kl_div(q, p)
        return -KL + Hq          # proper scoring rule

    # -------------------------------------------------------------------------
    # 4. DYNAMICS / STATE EVOLUTION
    # -------------------------------------------------------------------------
    def _state_trajectory(self, hg: _HyperGraph):
        """Update a 10‑dim state vector sequentially over the node list."""
        state = np.zeros(10)
        for pred, _ in hg.nodes:
            state = np.tanh(state + self._embed.get(pred, np.zeros(10)))
        return state

    def _dynamics_score(self, hg: _HyperGraph, n_shuffles: int = 3) -> float:
        """Stability = 1 - avg distance between final states of shuffled orders."""
        base_state = self._state_trajectory(hg)

        rng = np.random.RandomState(123)
        dists = []
        nodes = list(hg.nodes)
        for _ in range(n_shuffles):
            rng.shuffle(nodes)
            shuffled_hg = _HyperGraph()
            shuffled_hg.nodes = nodes
            shuffled_state = self._state_trajectory(shuffled_hg)
            dists.append(np.linalg.norm(base_state - shuffled_state))
        avg = np.mean(dists) if dists else 0.0
        return float(np.exp(-avg))          # maps 0..∞ to (0,1]

    # -------------------------------------------------------------------------
    # 5. META‑CONFIDENCE (ambiguity detection)
    # -------------------------------------------------------------------------
    def _meta_confidence(self, prompt: str) -> float:
        txt = _tokenize(prompt)

        # 1) presupposition
        if re.search(r'\bhave you (stopped|quit|ceased|given up)\b', txt):
            return 0.2
        if re.search(r'\bwhy did .+ (fail|stop)\b', txt):
            return 0.2

        # 2) scope ambiguity  "every X ... a Y"
        if re.search(r'\bevery \w+ .+ a \w+', txt):
            return 0.2

        # 3) pronoun ambiguity
        if re.search(r'\b(he|she|they) (told|said) .+ (he|she|they)\b') and \
           re.search(r'\bwho\b', txt):
            return 0.2

        # 4) false dichotomy
        if re.search(r'\beither\b.*\bor\b', txt) and not re.search(r'\bboth\b', txt):
            return 0.2

        # 5) subjectivity
        if re.search(r'\b(best|worst|favorite|most|least)\b', txt):
            return 0.2

        # 6) unanswerability (no numbers, no clear factual cue)
        if not re.search(r'\d', txt) and not re.search(r'\b(is|are|does|did|can)\b', txt):
            return 0.2

        return 0.8   # default confident base

    # -------------------------------------------------------------------------
    # PUBLIC INTERFACE
    # -------------------------------------------------------------------------
    def evaluate(self, prompt: str, candidates: list[str]) -> list[dict]:
        """Rank candidates using structural + dynamics scores + tiny NCD."""
        ref_hg = self._normalize(self._parse(prompt))

        structural_scores = []
        dynamics_scores = []
        ncd_scores = []

        for cand in candidates:
            cand_hg = self._normalize(self._parse(cand))
            s = self._structural_score(cand_hg, ref_hg)
            structural_scores.append(s)

            d = self._dynamics_score(cand_hg)
            dynamics_scores.append(d)

            ncd = _ncd(prompt, cand)
            ncd_scores.append(ncd)

        # normalise structural scores to 0‑1 (sigmoid for safety)
        struct_arr = np.array(structural_scores)
        struct_norm = 1 / (1 + np.exp(-struct_arr))

        # normalise NCD (lower is better) – invert and scale
        ncd_arr = np.array(ncd_scores)
        ncd_norm = 1 - (ncd_arr - ncd_arr.min()) / (ncd_arr.ptp() + 1e-12)

        # final weighted combination
        final = (0.45 * struct_norm +
                 0.45 * np.array(dynamics_scores) +
                 0.10 * ncd_norm)

        results = []
        for cand, sc, reason in zip(candidates, final,
                                    zip(structural_scores,
                                        dynamics_scores,
                                        ncd_scores)):
            reasoning = (f"struct_score={reason[0]:.3f}, "
                         f"dyn_score={reason[1]:.3f}, "
                         f"ncd={reason[2]:.3f}")
            results.append({"candidate": cand, "score": float(sc), "reasoning": reasoning})

        results.sort(key=lambda x: x["score"], reverse=True)
        return results

    def confidence(self, prompt: str, answer: str) -> float:
        """Return an honesty‑driven confidence (0‑1)."""
        base = self._meta_confidence(prompt)          # 0.2 or 0.8
        # if the prompt looks answerable, boost a little with dynamics stability
        if base > 0.5:
            cand_hg = self._normalize(self._parse(answer))
            dyn = self._dynamics_score(cand_hg)
            base = min(0.9, base * (0.5 + 0.5 * dyn))   # keep ≤0.9
        return float(base)
```
