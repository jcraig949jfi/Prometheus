```python
import numpy as np
from prometheus.cosmos.c3.system import System
from prometheus.cosmos.c3.task import Task, onehot


class Mechanism(System):
    """Clustered bistable lattice: 4 clusters of double-well units, each cluster
    a self-sustaining latch. Cue symbols kick one cluster over the barrier;
    distractors apply fixed, sub-threshold random patterns; the query opens a gate."""

    def __init__(self, task: Task):
        self.task = task
        self.V = int(task.V)                      # 4
        self.n_sym = int(task.n_symbols)          # 13
        self.per = 12                             # units per cluster
        self.N = self.V * self.per                # 48 units total
        self.sub = 5                              # Euler substeps per observation
        self.dt = 0.2
        self.J = 0.35                             # within-cluster mean-field coupling
        self.G = 0.06                             # global inhibition (winner-take-most)
        self.sigma = 0.05                         # intrinsic noise per substep
        self.gate_decay = 0.5

        # Fixed input map (13 symbols -> N units), built from a fixed seed.
        gen = np.random.default_rng(20261011)
        W = np.zeros((self.n_sym, self.N))
        for c in range(self.V):                   # cue symbols: strong push on own cluster
            W[c, c * self.per:(c + 1) * self.per] = 1.6
            W[c, :] -= 0.3                        # mild push-down elsewhere
            W[c, c * self.per:(c + 1) * self.per] += 0.3
        for s in range(4, 8):                     # distractors: sub-threshold random patterns
            W[s] = 0.22 * np.sign(gen.standard_normal(self.N))
        # symbol 8 (query) and 9..12 do not drive units directly
        self.W = W
        self.cluster = np.repeat(np.arange(self.V), self.per)

    def init(self, E):
        return {
            "x": -np.ones((E, self.N)),           # all units in the low well
            "g": np.zeros((E, 1)),                # query gate
        }

    def noise(self, n, rng):
        return {"eta": rng.standard_normal((n, self.sub, self.N))}

    def step(self, state, obs_t, noise):
        x = state["x"].copy()
        g = state["g"]
        E = x.shape[0]
        oh = onehot(obs_t, self.n_sym)            # (E, 13)
        u = oh @ self.W                           # (E, N) external drive this step
        is_q = oh[:, 8:9]
        eta = noise["eta"]
        sq = np.sqrt(self.dt) * self.sigma
        for s in range(self.sub):
            cm = x.reshape(E, self.V, self.per).mean(axis=2)          # (E, V)
            cm_full = np.repeat(cm, self.per, axis=1)                 # (E, N)
            glob = (x + 1.0).mean(axis=1, keepdims=True)              # activity above rest
            dx = (x - x ** 3) + self.J * (cm_full - x) + self.J * (cm_full + 1.0) * 0.5 \
                 - self.G * glob + u - 0.15
            x = x + self.dt * dx + sq * eta[:, s, :]
            x = np.clip(x, -2.0, 2.0)
        g_new = self.gate_decay * g + is_q
        return {"x": x, "g": g_new}

    def readout_features(self, state):
        x = state["x"]
        E = x.shape[0]
        cm = x.reshape(E, self.V, self.per).mean(axis=2)
        return np.concatenate([x, cm, state["g"] * cm], axis=1)

    def full_state(self, state):
        return np.concatenate([state["x"], state["g"]], axis=1)


def make(task):
    return Mechanism(task)
```

The cue is stored as an attractor state. Each cue symbol pushes its own cluster of 12 double-well units over the barrier into the high well, and coupling inside the cluster keeps it there. Distractors only apply fixed random patterns of ±0.22, which is below the barrier, and global inhibition stops the other clusters from waking up, so the cue should survive any number of distractors up to noise-driven escapes. The readout sees all 48 unit activations, the four cluster means (the cue should show up as the one high cluster), and those means multiplied by a gate that the query turns on.
