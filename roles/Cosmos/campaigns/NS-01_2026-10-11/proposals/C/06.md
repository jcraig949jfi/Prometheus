This one stores the cue in the hysteresis of a large set of two-state relays (return-point memory), rather than in a dynamical attractor.

```python
import numpy as np
from prometheus.cosmos.c3.system import System
from prometheus.cosmos.c3.task import Task, onehot


class Mechanism(System):
    def __init__(self, task: Task):
        self.task = task
        self.S = task.n_symbols
        self.R = 256                                  # hysteretic relays (Preisach ensemble)
        g = np.random.default_rng(9001)
        a = g.uniform(-2.5, 2.5, self.R)
        b = g.uniform(-2.5, 2.5, self.R)
        self.alpha = np.maximum(a, b)                 # up-switching threshold
        self.beta = np.minimum(a, b)                  # down-switching threshold
        H = np.zeros(self.S)
        H[0:4] = [0.6, 1.1, 1.6, 2.1]                 # cue: field excursion of increasing height
        H[4:8] = [-0.9, -0.4, 0.1, 0.5]               # distractors: smaller excursions, both signs
        H[8] = 0.0                                    # query: neutral field
        self.H = H
        self.sig_field = 0.15                         # common field noise
        self.sig_local = 0.12                         # per-relay (thermal) noise
        self.G = 16                                   # readout pick-up coils
        self.groups = g.integers(0, self.G, self.R)   # random coil assignment
        Wg = np.zeros((self.R, self.G))
        Wg[np.arange(self.R), self.groups] = 1.0
        self.Wg = Wg / np.maximum(Wg.sum(axis=0, keepdims=True), 1)

    def init(self, E):
        return {"s": -np.ones((E, self.R))}           # saturated down

    def noise(self, n, rng):
        return {"h": rng.standard_normal((n, 1)),
                "loc": rng.standard_normal((n, self.R))}

    def step(self, state, obs_t, noise):
        s = state["s"]
        field = (self.H[obs_t][:, None] + self.sig_field * noise["h"]
                 + self.sig_local * noise["loc"])
        s_new = np.where(field >= self.alpha[None, :], 1.0,
                         np.where(field <= self.beta[None, :], -1.0, s))
        return {"s": s_new}

    def readout_features(self, state):
        return state["s"] @ self.Wg

    def full_state(self, state):
        return state["s"].copy()


def make(task):
    return Mechanism(task)
```

The cue drives a field excursion to one of four heights, which flips up every relay whose up-switching threshold is below that height. Afterwards, relays whose thresholds lie above anything a distractor reaches, and whose down-switching thresholds lie below anything a distractor reaches, stay flipped. This is the wiping-out and return-point memory of a Preisach ensemble. Thermal noise on each relay's field occasionally flips relays near the edges of that region. The decoder sees magnetisation averaged over 16 randomly wired pick-up coils, so the cue is read as how far the remembered excursion reached, measured through mixed group averages.
